// --- SISTEMA DE UPLOAD COM PERSISTÊNCIA --- const fileInput = document.getElementById('fileInput'); // 1. Função que executa o código do plugin function executePlugin(code) { try { const script = document.createElement('script'); script.text = code; document.body.appendChild(script); } catch (err) { console.error("Erro ao executar plugin salvo:", err); } } // 2. Carregar plugins salvos assim que a página abrir window.addEventListener('load', () => { const savedPlugins = JSON.parse(localStorage.getItem('myGamePlugins') || '[]'); savedPlugins.forEach(pluginCode => { executePlugin(pluginCode); }); console.log(`${savedPlugins.length} plugins restaurados da memória.`); }); // 3. Salvar novo plugin ao fazer upload fileInput.addEventListener('change', function(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(event) { const pluginCode = event.target.result; // Executa agora executePlugin(pluginCode); // Salva para sempre no LocalStorage const savedPlugins = JSON.parse(localStorage.getItem('myGamePlugins') || '[]'); savedPlugins.push(pluginCode); localStorage.setItem('myGamePlugins', JSON.stringify(savedPlugins)); alert("Plugin integrado e salvo com sucesso!"); }; reader.readAsText(file); }); // FUNÇÃO EXTRA: Se quiser deletar tudo e começar do zero, digite no console: // localStorage.clear(); location.reload();
